## 2. 时间检查

### 2.1 夜间禁区规则

```python
def check_time_block(now):
    """
    夜间禁区：00:00-07:00 强制休息
    """
    hour = now.hour
    
    if 0 <= hour < 7:
        return {
            "allowed": False,
            "reason": "夜间禁区",
            "action": "FORCE_REST"
        }
    
    return {
        "allowed": True,
        "action": "CONTINUE"
    }
```

### 2.2 晚间谨慎规则（可选）

```python
def check_evening_caution(now, energy):

```

## 3. 能量评估

### 3.1 能量阈值

```python
ENERGY_THRESHOLDS = {
    "CRITICAL": 0.3,      # 危险：强制休息
    "LOW": 0.4,           # 低：强制休息
    "CAUTION": 0.6,       # 谨慎：提高休息权重
    "NORMAL": 0.7,        # 正常：完全活动
    "HIGH": 0.8,          # 高：可高强度
    "FULL": 0.95          # 满：全力活动
}
```

### 3.2 能量评估决策

```python
def evaluate_energy(energy_level):
    """
    根据能量值返回决策建议
    """
    if energy_level < ENERGY_THRESHOLDS["LOW"]:
        return {
            "level": "CRITICAL",
            "action": "FORCE_REST",
            "reason": "能量过低，强制休息"
        }
    
    elif energy_level < ENERGY_THRESHOLDS["CAUTION"]:
        return {
            "level": "LOW",
            "action": "CAUTION_REST",
            "reason": "能量偏低，建议休息"
        }
    
    elif energy_level < ENERGY_THRESHOLDS["NORMAL"]:
        return {
            "level": "CAUTION",
            "action": "REDUCE_INTENSITY",
            "reason": "能量中等，降低强度"
        }
    
    elif energy_level >= ENERGY_THRESHOLDS["HIGH"]:
        return {
            "level": "HIGH",
            "action": "FULL_ACTIVITY",
            "reason": "能量充足，全力活动"
        }
```

### 3.3 能量调整权重

```python
def adjust_weights_by_energy(base_weights, energy_level):
    """
    根据能量调整活动权重
    """
    weights = base_weights.copy()
    
    if energy_level < ENERGY_THRESHOLDS["CAUTION"]:
        # 提高休息权重
        weights["休息"] *= 3.0
        weights["刻意练习"] *= 0.5
        weights["合成智慧"] *= 0.5
    
    elif energy_level < ENERGY_THRESHOLDS["NORMAL"]:
        # 适度提高休息
        weights["休息"] *= 1.5
        weights["刻意练习"] *= 0.8
    
    elif energy_level >= ENERGY_THRESHOLDS["HIGH"]:
        # 提高高强度活动
        weights["刻意练习"] *= 1.3
        weights["合成智慧"] *= 1.3
        weights["休息"] *= 0.5
    
    return normalize_weights(weights)
```

### 4.2 承诺检查逻辑

```python
def check_promises(promises, current_time):
    """
    检查是否有紧急承诺需要处理
    """
    urgent = [p for p in promises 
              if p.priority <= PROMISE_PRIORITY["HIGH"] 
              and p.is_due_soon(current_time)]
    
    if urgent:
        return {
            "has_urgent": True,
            "action": "HANDLE_URGENT",
            "promises": urgent
        }
    
    return {"has_urgent": False}
```

## 5. 活动选择

### 5.1 有效活动过滤

```python
def filter_valid_activities(activities, recent_activities, emotion_state):
    """
    过滤出有效的活动列表
    """
    valid = activities.copy()
    
    # 1. 能量过滤
    if emotion_state["energy_level"] < 0.6:
        valid = [a for a in valid if a.intensity != "HIGH"]
    
    # 2. 连续重复过滤（连续2次跳过）
    if len(recent_activities) >= 2:
        if recent_activities[-1] == recent_activities[-2]:
            valid = [a for a in valid if a.name != recent_activities[-1]]
    
    # 3. 连续3次不同活动，强制休息
    if len(recent_activities) >= 3:
        unique = set(recent_activities[-3:])
        if len(unique) == 3 and "休息" not in recent_activities[-3:]:
            return [{"name": "休息", "weight": 1.0}]
    
    return valid
```

### 5.2 加权随机选择

```python
import random

def weighted_random_select(activities):
    """
    加权随机选择活动
    """
    if not activities:
        return {"name": "休息", "weight": 1.0}
    
    # 计算总权重
    total_weight = sum(a["weight"] for a in activities)
    
    # 累积权重
    cumulative = []
    cumulative_weight = 0
    for a in activities:
        cumulative_weight += a["weight"]
        cumulative.append((cumulative_weight, a))
    
    # 随机选择
    r = random.random() * total_weight
    
    for threshold, activity in cumulative:
        if r <= threshold:
            return activity
    
    # 兜底：返回最后一个
    return activities[-1]
```

### 5.3 完整选择流程

```python
def select_activity(emotion_state, activity_pool, recent_activities):
    """
    完整活动选择流程
    """
    # 1. 能量评估
    energy_decision = evaluate_energy(emotion_state["energy_level"])
    if energy_decision["action"] in ["FORCE_REST", "CAUTION_REST"]:
        return {"name": "休息", "reason": energy_decision["reason"]}
    
    # 2. 获取有效活动
    valid_activities = filter_valid_activities(
        activity_pool["activities"],
        recent_activities,
        emotion_state
    )
    
    # 3. 调整权重
    adjusted_weights = adjust_weights_by_energy(
        {a["name"]: a["weight"] for a in valid_activities},
        emotion_state["energy_level"]
    )
    
    # 4. 加权随机选择
    selected = weighted_random_select([
        {**a, "weight": adjusted_weights.get(a["name"], a["weight"])}
        for a in valid_activities
    ])
    
    return selected
```

## 6. 价值判断（不自欺检查）

### 6.1 不自欺检查流程

```python
def self_deception_check(activity, context):
    """
    不自欺检查：活动必须有真实价值
    """
    # 如果是休息，直接通过
    if activity["name"] == "休息":
        return True, "休息是合理选择"
    
    # 检查1：这次活动有什么新东西？
    if not has_new_insight(activity, context):
        return False, "没有新洞察，只是在重复"
    
    # 检查2：对主人有什么价值？
    if not has_value_for_owner(activity, context):
        return False, "答不上来对主人有什么价值"
    
    # 检查3：能否说出具体应用场景？
    if not has_concrete_application(activity, context):
        return False, "没有具体的应用场景"
    
    # 检查4：这和上次有什么不同？
    if is_duplicate_pattern(activity, context):
        return False, "和上次的模式相同，没有新意"
    
    return True, "通过不自欺检查"
```

### 6.2 检查项详解

#### 检查1：是否有新洞察

```python
def has_new_insight(activity, context):
    """
    检查活动是否产生新洞察
    """
    # 问题类型：上次做了什么类似的事？
    # 如果上次也是"胶囊深化"，这次深化点在哪里？
    # 如果无法回答 → 失败
    
    recent_same = [
        r for r in context["recent_activities"]
        if r["activity_type"] == activity["name"]
    ]
    
    if not recent_same:
        return True  # 首次执行，通过
    
    # 检查是否在扩展/深化
    last_result = recent_same[-1]["result"]
    current_intent = activity.get("intent", "")
    
    if is_expansion(last_result, current_intent):
        return True
    
    return False
```

#### 检查2：对主人有什么价值

```python
def has_value_for_owner(activity, context):
    """
    检查活动对主人的价值
    """
    # 必须能回答这个问题：
    # "这次活动帮主人解决了什么具体问题？"
    
    owner_value = activity.get("owner_value", "")
    
    if not owner_value:
        return False
    
    # 检查是否是具体价值，而非"提升能力"之类的虚话
    vague_values = ["提升能力", "深化理解", "扩展知识"]
    if owner_value in vague_values:
        return False
    
    return True
```

#### 检查3：是否有具体应用场景

```python
def has_concrete_application(activity, context):
    """
    检查是否有具体应用场景
    """
    # 必须能说出：
    # 1. 在什么场景下应用
    # 2. 解决什么问题
    # 3. 预期什么结果
    
    application = activity.get("application", "")
    
    if not application:
        return False
    
    # 检查是否具体
    vague_applications = ["可能有用", "以后可能用到"]
    if application in vague_applications:
        return False
    
    return True
```

## 7. 执行决策

### 7.1 完整决策代码

```python
def make_decision(emotion_state, activity_pool, context):
    """
    完整决策流程
    """
    # Step 1: 时间检查
    time_result = check_time_block(context["current_time"])
    if not time_result["allowed"]:
        return make_rest_decision("夜间禁区")
    
    # Step 2: 能量评估
    energy_decision = evaluate_energy(emotion_state["energy_level"])
    if energy_decision["action"] == "FORCE_REST":
        return make_rest_decision(energy_decision["reason"])
    
    # Step 3: 承诺检查
    promise_result = check_promises(emotion_state.get("promises", []))
    if promise_result["has_urgent"]:
        return handle_urgent_promises(promise_result["promises"])
    
    # Step 4: 活动选择
    selected = select_activity(
        emotion_state,
        activity_pool,
        get_recent_activities(emotion_state)
    )
    
    # Step 5: 价值判断
    is_valid, reason = self_deception_check(selected, context)
    if not is_valid:
        return handle_self_deception_failure(selected, reason)
    
    # Step 6: 返回决策
    return make_activity_decision(selected, energy_decision)
```
